class Main {
    public static void main(String[] args) {
        int[] arr = {2,0,1,2,1,0,1};
        int zero = 0;
        int one = 0;
        int two = 0;
        for(int i = 0; i < arr.length; i++) {
            if (arr[i]==0)
            zero++;
            else if(arr[i]==1)
            one++;
            else
            two++;
        }
        for(int i = 0; i < zero; i++)
            System.out.print("0 ");
        for(int i = 0; i < one; i++)
            System.out.print("1 ");
        for(int i = 0; i < two; i++)
            System.out.print("2 ");
    }
}
