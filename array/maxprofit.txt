class Main {
    public static void main(String[] args) {
int[] price = {7, 1, 5, 3, 6, 4};

int maxProfit = 0;

for(int i = 0; i < price.length; i++) {

    for(int j = i + 1; j < price.length; j++) {

        int profit = price[j] - price[i];

        if(profit > maxProfit) {
            maxProfit = profit;
        }
    }
}

System.out.println(maxProfit);}}
